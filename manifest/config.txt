# Project defaults. Override DOME_USER/DOME_TARGET in manifest/user.txt (gitignored).
ROS_DISTRO=kilted
UBUNTU_CODENAME=noble
DOME_USER=robot

# DOME_TARGET=pi|vm|cloud — pi installs Pi-hardware packages/repos/overlays;
# vm and cloud skip them. cloud additionally enables swap (like pi) for a
# small cloud shape; see 02-doc/cloud-howto.md.
DOME_TARGET=pi

# SWAP_SIZE_MB — swapfile size in MB for pi and cloud, guards against colcon
# build OOM. Not created for vm. 0 disables.
SWAP_SIZE_MB=2048

# DOME_MODE=native|docker — native skips Docker install/dome.service (Scenarios 1/2).
# Only Scenario 3 (docker-howto.md) sets DOME_MODE=docker in manifest/user.txt.
DOME_MODE=native

# DOME_DESKTOP=none|vnc — vnc installs an xfce/TigerVNC/noVNC desktop for
# tools needing a real X display (e.g. rviz2). Off by default; see cloud-howto.md.
DOME_DESKTOP=none

# DOME_VNC_ACCESS=none|public|tunnel — how the vnc desktop is reached (F14).
# Requires DOME_DESKTOP=vnc. none = no remote VNC; public = one TigerVNC served
# on 0.0.0.0:48210 (VNC password, set once with vncpasswd) reachable by URL,
# with the firewall opened by Terraform var vnc_access; tunnel = loopback only,
# reached over `ssh -L` by a user whose public key is on the box.
DOME_VNC_ACCESS=none

# REPOS_CONFIG — which named set of repos this host clones (F17). The value is
# a set name mapped to manifest/repos-<name>.txt, lowercased: DOME is the full
# robot set, SAFE is a minimal ROS 2 dev environment (rosutils + better_launch,
# all HTTPS, no GitHub credential needed). NONE is a reserved value meaning
# clone nothing; it maps to no file, so manifest/repos-none.txt must never be
# created. A name with no matching file is a hard error, never a fallback.
# Deliberately not DOME_-prefixed, unlike the globals above. Add a set by
# dropping in a new repos-<name>.txt — no code change.
REPOS_CONFIG=DOME
